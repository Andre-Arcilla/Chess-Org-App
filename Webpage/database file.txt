CREATE TABLE Chessistant.Announcements (
  AnnID bigint GENERATED ALWAYS AS IDENTITY NOT NULL CHECK ("AnnID" >= 0),
  Author text NOT NULL,
  LastEditor text NOT NULL,
  Title text NOT NULL DEFAULT 'Title'::text,
  Date timestamp with time zone NOT NULL DEFAULT now(),
  Text text NOT NULL DEFAULT 'Text'::text,
  LastModified bigint NOT NULL DEFAULT EXTRACT(epoch FROM now()),
  CONSTRAINT Announcements_pkey PRIMARY KEY (AnnID),
  CONSTRAINT fk_ann_author FOREIGN KEY (Author) REFERENCES Chessistant.Profiles(StudNum),
  CONSTRAINT fk_ann_editor FOREIGN KEY (LastEditor) REFERENCES Chessistant.Profiles(StudNum)
);
CREATE TABLE Chessistant.ChessGames (
  GameID bigint GENERATED ALWAYS AS IDENTITY NOT NULL CHECK ("GameID" >= 0),
  GameNum integer NOT NULL DEFAULT 1,
  StudNum text NOT NULL,
  PlayerColor text NOT NULL DEFAULT 'White'::text CHECK ("PlayerColor" = ANY (ARRAY['White'::text, 'Black'::text])),
  Date timestamp with time zone NOT NULL DEFAULT now(),
  Result text NOT NULL DEFAULT 'Draw'::text CHECK ("Result" = ANY (ARRAY['Win'::text, 'Lose'::text, 'Draw'::text])),
  PGN text NOT NULL DEFAULT 'N/A'::text,
  Feedback text,
  LastModified bigint NOT NULL DEFAULT EXTRACT(epoch FROM now()),
  FeedbackRead boolean NOT NULL DEFAULT false,
  CONSTRAINT ChessGames_pkey PRIMARY KEY (GameID),
  CONSTRAINT fk_chess_student FOREIGN KEY (StudNum) REFERENCES Chessistant.Profiles(StudNum)
);
CREATE TABLE Chessistant.OrgRoster (
  MmbrID bigint GENERATED ALWAYS AS IDENTITY NOT NULL CHECK ("MmbrID" >= 0),
  StudName text NOT NULL DEFAULT 'John Smith'::text,
  StudNum text NOT NULL UNIQUE,
  LastModified bigint NOT NULL DEFAULT EXTRACT(epoch FROM now()),
  CONSTRAINT OrgRoster_pkey PRIMARY KEY (MmbrID)
);
CREATE TABLE Chessistant.Profiles (
  UserID bigint GENERATED ALWAYS AS IDENTITY NOT NULL CHECK ("UserID" >= 0),
  StudName text NOT NULL DEFAULT 'John Smith'::text,
  StudNum text NOT NULL UNIQUE,
  Email text NOT NULL DEFAULT 'no.mail@umak.edu.ph'::text,
  Password text NOT NULL DEFAULT '12345'::text,
  Rating integer NOT NULL DEFAULT 100 CHECK ("Rating" >= 1),
  PuzzlesWin integer NOT NULL DEFAULT 0,
  PuzzlesTotal integer NOT NULL DEFAULT 0,
  Role text NOT NULL DEFAULT 'Member'::text CHECK ("Role" = ANY (ARRAY['Member'::text, 'Coach'::text, 'Admin'::text, 'Disabled'::text])),
  Date timestamp with time zone NOT NULL DEFAULT now(),
  LastModified bigint NOT NULL DEFAULT EXTRACT(epoch FROM now()),
  CONSTRAINT Profiles_pkey PRIMARY KEY (UserID)
);
CREATE TABLE Chessistant.Registrations (
  RegID bigint GENERATED ALWAYS AS IDENTITY NOT NULL CHECK ("RegID" >= 0),
  StudName text NOT NULL DEFAULT 'John Smith'::text,
  StudNum text NOT NULL UNIQUE,
  Email text NOT NULL DEFAULT 'no.mail@umak.edu.ph'::text,
  Password text NOT NULL DEFAULT '12345'::text,
  Date timestamp with time zone NOT NULL DEFAULT now(),
  LastModified bigint NOT NULL DEFAULT EXTRACT(epoch FROM now()),
  CONSTRAINT Registrations_pkey PRIMARY KEY (RegID)
);
CREATE TABLE Chessistant.TournamentMatches (
  matchid bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  tourid bigint NOT NULL,
  player1 text NOT NULL,
  player2 text NOT NULL,
  player1result numeric NOT NULL DEFAULT 0.5,
  player2result numeric NOT NULL DEFAULT 0.5,
  CONSTRAINT TournamentMatches_pkey PRIMARY KEY (matchid),
  CONSTRAINT tournamentmatches_tourid_fkey FOREIGN KEY (tourid) REFERENCES Chessistant.Tournaments(TourID),
  CONSTRAINT TournamentMatches_player1_fkey FOREIGN KEY (player1) REFERENCES Chessistant.Profiles(StudNum),
  CONSTRAINT TournamentMatches_player2_fkey FOREIGN KEY (player2) REFERENCES Chessistant.Profiles(StudNum)
);
CREATE TABLE Chessistant.TournamentParticipants (
  TPID bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  TourID bigint NOT NULL,
  StudNum text NOT NULL,
  CONSTRAINT TournamentParticipants_pkey PRIMARY KEY (TPID),
  CONSTRAINT TournamentParticipants_TourID_fkey FOREIGN KEY (TourID) REFERENCES Chessistant.Tournaments(TourID),
  CONSTRAINT TournamentParticipants_StudNum_fkey FOREIGN KEY (StudNum) REFERENCES Chessistant.Profiles(StudNum)
);
CREATE TABLE Chessistant.Tournaments (
  TourID bigint GENERATED ALWAYS AS IDENTITY NOT NULL CHECK ("TourID" >= 0),
  Author text NOT NULL,
  LastEditor text NOT NULL,
  Title text NOT NULL DEFAULT 'Title'::text,
  Date timestamp with time zone NOT NULL DEFAULT now(),
  Text text NOT NULL DEFAULT 'Text'::text,
  LastModified bigint NOT NULL DEFAULT EXTRACT(epoch FROM now()),
  Style text NOT NULL DEFAULT 'Round Robin'::text,
  ParticipantCount integer NOT NULL DEFAULT 6,
  CONSTRAINT Tournaments_pkey PRIMARY KEY (TourID),
  CONSTRAINT fk_tour_author FOREIGN KEY (Author) REFERENCES Chessistant.Profiles(StudNum),
  CONSTRAINT fk_tour_editor FOREIGN KEY (LastEditor) REFERENCES Chessistant.Profiles(StudNum)
);
